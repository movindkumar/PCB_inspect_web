const MAT = {
  pcb:      new THREE.MeshStandardMaterial({ color:0x0d4a14, roughness:0.55, metalness:0.1 }),
  mask:     new THREE.MeshStandardMaterial({ color:0x0b4010, roughness:0.45, metalness:0.05, transparent:true, opacity:0.93 }),
  copper:   new THREE.MeshStandardMaterial({ color:0xb87333, roughness:0.30, metalness:0.90 }),
  pad:      new THREE.MeshStandardMaterial({ color:0xd4924a, roughness:0.25, metalness:0.92 }),

// ── Geometry utils ─────────────────────────────────────────────────────────────
function box(w,h,d,mat,x,y,z,rx,ry,rz){
  const m=new THREE.Mesh(new THREE.BoxGeometry(w,h,d),mat);
  m.position.set(x,y,z);
  if(rx)m.rotation.x=rx; if(ry)m.rotation.y=ry; if(rz)m.rotation.z=rz;
  m.castShadow=true; m.receiveShadow=true;
  return m;
}
function cyl(rt,rb,h,seg,mat,x,y,z){
  const m=new THREE.Mesh(new THREE.CylinderGeometry(rt,rb,h,seg),mat);
  m.position.set(x,y,z); m.castShadow=true; return m;
}
function smtComp(g,x,z,w,l){
  // SMT resistor/capacitor
  g.add(box(w,0.08,l,MAT.smt,x,0.09,z));
  g.add(box(0.06,0.08,l*0.7,MAT.copper,x-w/2,0.09,z));
  g.add(box(0.06,0.08,l*0.7,MAT.copper,x+w/2,0.09,z));
}
function trace(g,x,z,len,w,dir,thick=0.055){
  const gw=dir==='x'?len:w, gd=dir==='x'?w:len;
  g.add(box(gw,thick,gd,MAT.copper,x,thick/2,z));
}
function padHole(g,x,z,pr=0.3,dr=0.15){
  g.add(cyl(pr,pr,0.07,24,MAT.pad,x,0.035,z));
  g.add(cyl(dr,dr,0.45,24,MAT.hole,x,0,z));
}
function addFloor(g){
  const fl=new THREE.Mesh(new THREE.PlaneGeometry(30,30),
    new THREE.MeshStandardMaterial({color:0x050a05,roughness:1}));
  fl.rotation.x=-Math.PI/2; fl.position.y=-0.5; fl.receiveShadow=true;
  g.add(fl);
}

// DEFECT 1 — Missing Hole
// ═══════════════════════════════════════════════════════════════════════════════
function buildMissingHole(){
  const g=new THREE.Group();
  addFloor(g);

  // PCB substrate
  g.add(box(10,0.38,7,MAT.pcb,0,-0.19,0));
  // Soldermask
  g.add(box(10,0.05,7,MAT.mask,0,0.025,0));

  // Copper traces grid
  for(let i=-2;i<=2;i++) trace(g,0,i*1.1,8.5,0.11,'x');
  for(let i=-3;i<=3;i++) trace(g,i*1.2,0,5.5,0.09,'z');

  // IC chip (large golden die)
  g.add(box(2.2,0.18,2.2,MAT.chip,1.5,0.09,0));
  g.add(box(1.8,0.08,1.8,MAT.chipGold,1.5,0.22,0));
  // IC pins
  for(let i=-3;i<=3;i++){
    g.add(box(0.55,0.06,0.08,MAT.copper,0.45+i*0.06*2,0.03,1.1));
    g.add(box(0.55,0.06,0.08,MAT.copper,0.45+i*0.06*2,0.03,-1.1));
  }

  // SMT components
  smtComp(g,-2.5,1.5,0.28,0.5);
  smtComp(g,-2.5,-1.5,0.28,0.5);
  smtComp(g,-1.5,2.2,0.18,0.35);
  smtComp(g,-3.5,0.5,0.18,0.35);

  // Via/pad grid — 3×4 array, one MISSING
  const padPositions=[
    [-3.8,-2.5],[-3.8,-1.2],[-3.8,0.1],[-3.8,1.4],
    [-2.8,-2.5],[-2.8,-1.2],[-2.8,0.1],[-2.8,1.4],
    [-1.8,-2.5],[-1.8,-1.2],[-1.8,0.1],[-1.8,1.4],
  ];
  const missingIdx=5; // [-2.8,-1.2] — second row, second col

  padPositions.forEach(([x,z],i)=>{
    if(i===missingIdx){
      // MISSING HOLE — solid copper pad, NO drill
      g.add(cyl(0.30,0.30,0.07,32,MAT.pad,x,0.035,z));
      // Dark fill centre (no hole)
      g.add(cyl(0.18,0.18,0.08,32,
        new THREE.MeshStandardMaterial({color:0x0d3b0d,roughness:0.8}),
        x,0.04,z));
      // Red indicator ring
      const ring=new THREE.Mesh(new THREE.TorusGeometry(0.34,0.06,10,32),MAT.defRed);
      ring.rotation.x=Math.PI/2; ring.position.set(x,0.12,z); g.add(ring);
      // Pulsing glow sphere
      const glow=new THREE.Mesh(new THREE.SphereGeometry(0.15,16,16),
        new THREE.MeshStandardMaterial({color:0xff1744,emissive:0xff1744,
          emissiveIntensity:1.5,transparent:true,opacity:0.7}));
      glow.position.set(x,0.55,z); glow.userData.pulse=true; g.add(glow);
    } else {
      padHole(g,x,z);
    }
  });

  return g;
}

// DEFECT 2 — Mouse Bite  (realistic edge notches like the photo)
// ═══════════════════════════════════════════════════════════════════════════════
function buildMouseBite(){
  const g=new THREE.Group();
  addFloor(g);

  const BW=10, BH=7, THICK=0.38;

  // Build PCB shape with semicircular mouse-bite notches on the LEFT edge (x = -BW/2)
  // Notches are punched in during depaneling — they look exactly like the photo edge
  const notchData=[
    // [z-centre, radius]
    [-2.8, 0.40],
    [-1.6, 0.36],
    [-0.4, 0.42],
    [ 0.8, 0.38],
    [ 2.0, 0.41],
    [ 3.0, 0.35],
  ];

  // Build shape in XZ plane (extruded along Y)
  const shape=new THREE.Shape();
  const x0=-BW/2, x1=BW/2, z0=-BH/2, z1=BH/2;

  // Start bottom-right, go clockwise
  shape.moveTo(x1, z0);
  shape.lineTo(x1, z1);   // right edge (clean)
  shape.lineTo(x0, z1);   // top edge (clean)

  // LEFT edge — walk from z1 down to z0, punching semicircle notches
  // Sort notches top→bottom
  const sorted=[...notchData].sort((a,b)=>b[0]-a[0]);
  let curZ=z1;
  sorted.forEach(([cz,r])=>{
    const top=cz+r, bot=cz-r;
    if(top<curZ) shape.lineTo(x0,top);
    // Semicircle arc going INTO the board (positive x direction)
    shape.absarc(x0, cz, r, Math.PI/2, -Math.PI/2, true);
    curZ=bot;
  });
  shape.lineTo(x0, z0);   // remainder of left edge
  shape.lineTo(x1, z0);   // bottom edge (clean)

  const extSettings={ depth:THICK, bevelEnabled:false };
  const boardGeo=new THREE.ExtrudeGeometry(shape, extSettings);

  // Rotate so board lies flat (shape is in XZ, extrude goes along Y, we want board flat)
  const board=new THREE.Mesh(boardGeo, MAT.pcb);
  board.rotation.x=-Math.PI/2;
  board.position.y=-THICK;
  board.receiveShadow=true; board.castShadow=true;
  g.add(board);

  // Soldermask as a clipped flat plane — approximate with same shape slightly thinner
  const maskShape=new THREE.Shape();
  maskShape.moveTo(x1,z0); maskShape.lineTo(x1,z1); maskShape.lineTo(x0,z1);
  let mCurZ=z1;
  sorted.forEach(([cz,r])=>{
    const top=cz+r, bot=cz-r;
    if(top<mCurZ) maskShape.lineTo(x0,top);
    maskShape.absarc(x0,cz,r,Math.PI/2,-Math.PI/2,true);
    mCurZ=bot;
  });
  maskShape.lineTo(x0,z0); maskShape.lineTo(x1,z0);
  const maskGeo=new THREE.ShapeGeometry(maskShape);
  const maskMesh=new THREE.Mesh(maskGeo,MAT.mask);
  maskMesh.rotation.x=-Math.PI/2;
  maskMesh.position.y=0.025;
  g.add(maskMesh);

  // Copper traces — horizontal lines across the board
  for(let i=-2;i<=2;i++) trace(g,0,i*1.1,8.0,0.12,'x');
  for(let i=-2;i<=2;i++) trace(g,i*1.4,0,5.5,0.09,'z');

  // Some traces get CLIPPED at the left edge (they'd connect to adjacent PCB in panel)
  // show them ending abruptly at the notch area
  const clippedTraces=[[-1.6,-0.8],[-0.4,1.1],[0.8,-1.8]];
  clippedTraces.forEach(([z,xEnd])=>{
    trace(g,(x0+xEnd)/2+0.5,z,Math.abs(xEnd-x0),0.12,'x');
  });

  // SMT components
  smtComp(g, 1.0, 2.0, 0.28, 0.5);
  smtComp(g, 3.0,-1.0, 0.28, 0.5);
  smtComp(g,-0.5,-2.0, 0.18, 0.35);
  smtComp(g, 2.5, 0.5, 0.18, 0.35);
  smtComp(g,-1.5, 1.0, 0.18, 0.35);

  // Via holes scattered
  [[2,-2],[3,2],[-1,2.5],[4,-1],[0,0]].forEach(([x,z])=>padHole(g,x,z,0.22,0.12));

  // IC chip
  g.add(box(2.0,0.18,2.0,MAT.chip, 2.5,0.09,-0.5));
  g.add(box(1.6,0.08,1.6,MAT.chipGold,2.5,0.22,-0.5));

  // Highlight each notch with amber glow
  sorted.forEach(([cz,r])=>{
    // Amber glow arc mesh (torus arc approximation — partial torus)
    const torusGeo=new THREE.TorusGeometry(r+0.04,0.07,8,20,Math.PI);
    const glow=new THREE.Mesh(torusGeo,MAT.defAmber);
    glow.rotation.y=Math.PI/2;  // face outward from left edge
    glow.position.set(x0+0.02,0.05,cz);
    glow.userData.pulse=true;
    g.add(glow);

    // Tiny sphere dot at deepest bite point
    const dot=new THREE.Mesh(new THREE.SphereGeometry(0.1,10,10),
      new THREE.MeshStandardMaterial({color:0xffd54f,emissive:0xff8f00,emissiveIntensity:2.0}));
    dot.position.set(x0+r+0.05,0.15,cz);
    dot.userData.pulse=true;
    g.add(dot);
  });

  // Callout label arrow pointing at notch region
  const arrowMat=new THREE.MeshStandardMaterial({color:0xffa000,emissive:0xffa000,emissiveIntensity:0.8});
  g.add(box(0.05,0.05,1.8,arrowMat,-5.5,0.8,0));
  g.add(box(0.8,0.05,0.05,arrowMat,-5.1,0.8,0));

  return g;
}

// DEFECT 3 — Open Circuit
// ═══════════════════════════════════════════════════════════════════════════════
function buildOpenCircuit(){
  const g=new THREE.Group();
  addFloor(g);

  g.add(box(10,0.38,7,MAT.pcb,0,-0.19,0));
  g.add(box(10,0.05,7,MAT.mask,0,0.025,0));

  // Background traces
  for(let i=-2;i<=2;i++) trace(g,0,i*1.0,8.5,0.10,'x');
  for(let i=-3;i<=3;i++) trace(g,i*1.0,0,5.5,0.08,'z');

  // 3 broken traces (left segment, gap, right segment)
  const breaks=[[-2,0.55],[0,0.55],[2,0.55]];
  breaks.forEach(([z,gx])=>{
    // Left segment
    g.add(box(3.2,0.06,0.14,MAT.copper,-3.4,0.03,z));
    // Right segment
    g.add(box(3.2,0.06,0.14,MAT.copper, 3.4,0.03,z));

    // Gap (dark substrate visible)
    g.add(box(gx*2,0.07,0.16,
      new THREE.MeshStandardMaterial({color:0x061006,roughness:1}),
      0,0.035,z));

    // Blue glow in gap
    g.add(box(gx*1.8,0.10,0.20,MAT.defBlue,0,0.05,z));

    // Spark dots at break ends
    [-gx,gx].forEach(ox=>{
      const dot=new THREE.Mesh(new THREE.SphereGeometry(0.09,10,10),
        new THREE.MeshStandardMaterial({color:0x90caf9,emissive:0x42a5f5,emissiveIntensity:2.5}));
      dot.position.set(ox,0.18,z);
      dot.userData.pulse=true;
      g.add(dot);
    });
  });

  // Component pads at ends
  [[-2],[0],[2]].forEach(([z])=>{
    padHole(g,-4.5,z,0.28,0.13);
    padHole(g, 4.5,z,0.28,0.13);
  });

  // Vertical connector trace
  trace(g,-4.5,0,4.2,0.10,'z');
  trace(g, 4.5,0,4.2,0.10,'z');

  // SMT passives
  smtComp(g,-1.5, 2.5,0.24,0.45);
  smtComp(g, 2.0,-2.5,0.24,0.45);
  smtComp(g, 3.5, 1.8,0.18,0.32);

  // Small IC
  g.add(box(1.6,0.15,1.2,MAT.chip,-2.5,0.075,-2.0));
  g.add(box(1.2,0.06,0.8,MAT.chipGold,-2.5,0.18,-2.0));

  return g;
}

// ── Scene management ──────────────────────────────────────────────────────────
let currentGroup=null;

function clearScene(){
  if(!currentGroup)return;
  scene.remove(currentGroup);
  currentGroup.traverse(o=>{ if(o.geometry)o.geometry.dispose(); });
  currentGroup=null;
}

function loadDefect(type){
  clearScene();
  if(type==='missing_hole')    currentGroup=buildMissingHole();
  else if(type==='mouse_bite') currentGroup=buildMouseBite();
  else                         currentGroup=buildOpenCircuit();
  scene.add(currentGroup);
  rotX=0.45; rotY=0.35; panX=0; panZ=0; zoom=11;
}

function switchDefect(type,btn){
  document.querySelectorAll('.tab').forEach(t=>t.classList.remove('active'));
  btn.classList.add('active');
  const info=INFO[type];
  document.getElementById('d-name').textContent=info.name;
  document.getElementById('d-desc').textContent=info.desc;
  const badge=document.getElementById('d-badge');
  badge.textContent=info.badge;
  badge.className='defect-badge '+info.badgeClass;
  loadDefect(type);
}

loadDefect('missing_hole');

let isDragging=false, isRight=false, lastX=0, lastY=0;
let rotX=0.45, rotY=0.35, panX=0, panZ=0, zoom=11;

canvas.addEventListener('mousedown',e=>{
  isDragging=true; isRight=e.button===2;
  lastX=e.clientX; lastY=e.clientY;
});
canvas.addEventListener('contextmenu',e=>e.preventDefault());
window.addEventListener('mouseup',()=>{ isDragging=false; });
window.addEventListener('mousemove',e=>{
  if(!isDragging)return;
  const dx=e.clientX-lastX, dy=e.clientY-lastY;
  if(isRight){ panX-=dx*0.012; panZ-=dy*0.012; }
  else {
    rotY+=dx*0.008; rotX+=dy*0.006;
    rotX=Math.max(-1.3,Math.min(1.3,rotX));
  }
  lastX=e.clientX; lastY=e.clientY;
});
canvas.addEventListener('wheel',e=>{
  zoom+=e.deltaY*0.02; zoom=Math.max(4,Math.min(24,zoom));
},{ passive:true });

// Touch
let lastTD=0;
canvas.addEventListener('touchstart',e=>{
  if(e.touches.length===1){
    isDragging=true; isRight=false;
    lastX=e.touches[0].clientX; lastY=e.touches[0].clientY;
  } else if(e.touches.length===2){
    const d=Math.hypot(e.touches[0].clientX-e.touches[1].clientX,
                       e.touches[0].clientY-e.touches[1].clientY);
    lastTD=d;
  }
});
canvas.addEventListener('touchmove',e=>{
  e.preventDefault();
  if(e.touches.length===1&&isDragging){
    const dx=e.touches[0].clientX-lastX, dy=e.touches[0].clientY-lastY;
    rotY+=dx*0.008; rotX+=dy*0.006;
    rotX=Math.max(-1.3,Math.min(1.3,rotX));
    lastX=e.touches[0].clientX; lastY=e.touches[0].clientY;
  } else if(e.touches.length===2){
    const d=Math.hypot(e.touches[0].clientX-e.touches[1].clientX,
                       e.touches[0].clientY-e.touches[1].clientY);
    zoom-=(d-lastTD)*0.05; zoom=Math.max(4,Math.min(24,zoom));
    lastTD=d;
  }
},{ passive:false });
canvas.addEventListener('touchend',()=>{ isDragging=false; });

// ── Render loop ───────────────────────────────────────────────────────────────
let t=0;
function animate(){
  requestAnimationFrame(animate);
  t+=0.016;
  if(!isDragging) rotY+=0.004;

  // Pulse defect indicators
  if(currentGroup){
    currentGroup.traverse(o=>{
      if(o.isMesh && o.userData.pulse && o.material.emissiveIntensity!==undefined){
        o.material.emissiveIntensity=0.8+Math.sin(t*3)*0.6;
      }
    });
  }

  const cx=Math.sin(rotY)*Math.cos(rotX)*zoom+panX;
  const cy=Math.sin(rotX)*zoom+1;
  const cz=Math.cos(rotY)*Math.cos(rotX)*zoom+panZ;
  camera.position.set(cx,cy,cz);
  camera.lookAt(panX,0,panZ);
  renderer.render(scene,camera);
}
animate();
